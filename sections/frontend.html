middleware.ts — Next.js route protection 
// middleware.ts (project root — runs before every page request)
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const PUBLIC_PATHS = ['/login', '/api/auth'];

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Allow public paths through
  if (PUBLIC_PATHS.some(p => pathname.startsWith(p))) {
    return NextResponse.next();
  }

  // Check for token in localStorage is not possible in middleware
  // so we rely on AuthProvider to redirect unauthenticated users
  return NextResponse.next();
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
};

providers/AuthProvider.tsx — where useAuth is declared 
'use client';
import { createContext, useContext, useEffect, useState } from 'react';
import { httpClient } from '../lib/httpClient';

interface AuthUser { sub: string; email: string; name: string; roles: string[]; }
interface AuthCtx  { user: AuthUser | null; token: string | null; isLoading: boolean; logout: () => void; }

const AuthContext = createContext<AuthCtx>({ user: null, token: null, isLoading: true, logout: () => {} });

// ── useAuth hook (declared here, used anywhere in the app) ──
export function useAuth() { return useContext(AuthContext); }

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser]       = useState<AuthUser | null>(null);
  const [token, setToken]     = useState<string | null>(null);
  const [isLoading, setLoading] = useState(true);

  useEffect(() => {
    const stored = localStorage.getItem('sema_token');
    if (!stored) { setLoading(false); return; }
    httpClient.get('/auth/session')
      .then(({ data }) => { setUser(data.user); setToken(data.token); })
      .catch(() => localStorage.removeItem('sema_token'))
      .finally(() => setLoading(false));
  }, []);

  function logout() {
    localStorage.removeItem('sema_token');
    setUser(null); setToken(null);
    window.location.href = `${process.env.NEXT_PUBLIC_SEMA_SSO_URL}/auth/logout`;
  }

  return (
    <AuthContext.Provider value={{ user, token, isLoading, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

lib/httpClient.ts — auto-attach token + refresh on 401 
import axios from 'axios';

export const httpClient = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL,
});

// Attach Bearer token from localStorage on every request
httpClient.interceptors.request.use(config => {
  const token = localStorage.getItem('sema_token');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

// On 401 → try to refresh token, then retry original request once
httpClient.interceptors.response.use(
  res => res,
  async err => {
    const original = err.config;
    if (err.response?.status === 401 && !original._retry) {
      original._retry = true;
      const { data } = await axios.post(
        `${process.env.NEXT_PUBLIC_SEMA_SSO_URL}/auth/refresh-token`,
        {}, { withCredentials: true }
      );
      localStorage.setItem('sema_token', data.token);
      original.headers.Authorization = `Bearer ${data.token}`;
      return httpClient(original);
    }
    return Promise.reject(err);
  }
);

app/login/page.tsx — redirect to SSO + receive token back 
'use client';
import { useEffect } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';

export default function LoginPage() {
  const router = useRouter();
  const params = useSearchParams();

  useEffect(() => {
    // Step 1: Gateway returns here with ?token=<jwt> after login
    const token = params.get('token');
    if (token) {
      localStorage.setItem('sema_token', token);
      router.replace('/dashboard');  // clean the token from the URL
      return;
    }

    // Step 2: No token → redirect to SSO gateway login
    // redirect_uri tells the gateway where to send the user back
    // callback_mode=direct tells it to append ?token= to the redirect_uri
    const ssoUrl   = process.env.NEXT_PUBLIC_SEMA_SSO_URL;
    const redirect = encodeURIComponent(`${window.location.origin}/login`);
    window.location.href = `${ssoUrl}/auth/login?redirect_uri=${redirect}&callback_mode=direct`;
  }, []);

  return <p>Redirecting to login…</p>;
}

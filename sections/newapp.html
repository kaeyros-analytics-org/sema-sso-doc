New App Integration Guide

Step-by-step: connect a brand new app to Sema SSO in under 30 minutes.

For a New Frontend (Next.js)
1
Set environment variables
.env.local
NEXT_PUBLIC_SEMA_SSO_URL=http://localhost:3001
NEXT_PUBLIC_API_URL=http://localhost:3003  # your backend port
2
Copy AuthProvider + useAuth from an existing app
Files to copy
cp sema-core-frontend/providers/AuthProvider.tsx  your-app/providers/
cp sema-core-frontend/hooks/useAuth.ts            your-app/hooks/
cp sema-core-frontend/lib/httpClient.ts           your-app/lib/
3
Wrap root layout with <AuthProvider>
app/layout.tsx
import { AuthProvider } from '../providers/AuthProvider';

export default function RootLayout({ children }) {
  return (
    <html><body>
      <AuthProvider>
        {children}
      </AuthProvider>
    </body></html>
  );
}
4
Create login page — receive token from gateway
app/login/page.tsx
'use client';
import { useEffect } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';

export default function LoginPage() {
  const router = useRouter();
  const params = useSearchParams();

  useEffect(() => {
    // Gateway returns here with ?token=<jwt>
    const token = params.get('token');
    if (token) {
      localStorage.setItem('sema_token', token);
      router.replace('/dashboard');
      return;
    }
    // Redirect to SSO — tell it to come back here with token
    const sso = process.env.NEXT_PUBLIC_SEMA_SSO_URL;
    const uri = encodeURIComponent(window.location.origin + '/login');
    window.location.href = `${sso}/auth/login?redirect_uri=${uri}&callback_mode=direct`;
  }, []);

  return <p>Redirecting…</p>;
}
5
Use httpClient for all API calls
Any component
import { httpClient } from '../lib/httpClient';

// Bearer token attached automatically, 401 triggers refresh automatically
const { data } = await httpClient.get('/api/your-endpoint', {
  headers: { 'x-credential-id': selectedCredentialId }
});
For a New Backend (NestJS)
1
Add environment variables
.env
SERVICE_ID=my-new-service              # unique name for this service
GATEWAY_TOKEN_SECRET=                  # copy from gateway .env
GATEWAY_SHARED_SECRET=                 # copy from gateway .env
SSO_API_GATEWAY_URL=http://localhost:3001
REDIS_HOST=localhost
REDIS_PORT=6379
DATABASE_URL=postgresql://postgres:1234567@localhost:5432/my-service
2
Install dependencies
npm
npm install @nestjs/jwt @nestjs/passport passport passport-jwt \
            @nestjs/config ioredis @liaoliaots/nestjs-redis
3
Copy JwtStrategy + GatewayMiddleware
Files to copy
cp sema-core-service/src/auth/jwt.strategy.ts      your-service/src/auth/
cp sema-core-service/src/auth/jwt-auth.guard.ts    your-service/src/auth/
cp sema-core-service/src/common/middleware/gateway.middleware.ts \
   your-service/src/common/middleware/
4
Register in AppModule
app.module.ts
@Module({
  imports: [
    ConfigModule.forRoot({ isGlobal: true }),
    JwtModule.register({ secret: process.env.GATEWAY_TOKEN_SECRET }),
    PassportModule,
  ],
  providers: [JwtStrategy],
})
export class AppModule implements NestModule {
  configure(consumer: MiddlewareConsumer) {
    consumer.apply(GatewayMiddleware).forRoutes('*');
  }
}
5
Protect routes with @UseGuards(JwtAuthGuard)
any.controller.ts
@Get('data')
@UseGuards(JwtAuthGuard)
async getData(@Headers('x-credential-id') credId: string, @Req() req) {
  // req.user = JWT payload (sub, email, roles, jti)
  // credId = scopes request to one company's data
  return this.service.getData(credId);
}
6
Add your service to the gateway route files
The gateway will return 404 for every request until your service is declared in all four route files. Open SEMA-SSO-api-gateway and add one entry in each file.
routes.yml (local) + .github/routes/routes-dev/stg/prod.yml
  # Add this block in each of the 4 route files
  # (adjust target URL per environment)
  - path: "my-new-service"        # must equal SERVICE_ID in your .env
    target: "http://localhost:3005"
    rewritePath: true

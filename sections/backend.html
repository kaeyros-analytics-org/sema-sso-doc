JWT Validation Strategy

Every NestJS backend implements a JwtStrategy (Passport.js) that validates:

app.module.ts — wire JwtModule + apply GatewayMiddleware globally 
import { Module, NestModule, MiddlewareConsumer } from '@nestjs/common';
import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { JwtStrategy } from './auth/jwt.strategy';
import { GatewayMiddleware } from './common/middleware/gateway.middleware';

@Module({
  imports: [
    PassportModule,
    JwtModule.register({
      secret: process.env.GATEWAY_TOKEN_SECRET, // must match gateway
      signOptions: { expiresIn: '1h' },
    }),
  ],
  providers: [JwtStrategy],
})
export class AppModule implements NestModule {
  configure(consumer: MiddlewareConsumer) {
    // Apply HMAC verification to ALL routes
    consumer.apply(GatewayMiddleware).forRoutes('*');
  }
}

auth/jwt.strategy.ts — full implementation 
import { Injectable, UnauthorizedException } from '@nestjs/common';
import { PassportStrategy } from '@nestjs/passport';
import { ExtractJwt, Strategy } from 'passport-jwt';
import { ConfigService } from '@nestjs/config';
import { InjectRedis } from '@liaoliaots/nestjs-redis';
import { Redis } from 'ioredis';

@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor(
    private config: ConfigService,
    @InjectRedis() private redis: Redis,
  ) {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      secretOrKey: config.get('GATEWAY_TOKEN_SECRET'),
    });
  }

  async validate(payload: any) {
    // 1. Verify this token was issued by the gateway (not Keycloak directly)
    if (payload.iss !== 'api-gateway') {
      throw new UnauthorizedException('Invalid token issuer');
    }

    // 2. Verify the audience includes this service
    const serviceId = this.config.get('SERVICE_ID'); // e.g. "sema-core-service"
    if (!payload.aud?.includes(serviceId)) {
      throw new UnauthorizedException('Token not intended for this service');
    }

    // 3. Check JTI blacklist (set by gateway on logout)
    const blacklisted = await this.redis.get(`blacklist:${payload.jti}`);
    if (blacklisted) {
      throw new UnauthorizedException('Token has been revoked');
    }

    return payload; // becomes req.user
  }
}
Credential Header

Multi-tenancy is implemented via the x-credential-id header.

Controller — reading x-credential-id for multi-tenant scoping 
import { Controller, Get, Headers, Request, UseGuards } from '@nestjs/common';
import { JwtAuthGuard } from '../auth/jwt-auth.guard';

@Controller('conversations')
export class ConversationController {
  constructor(private readonly service: ConversationService) {}

  @Get()
  @UseGuards(JwtAuthGuard)      // validates JWT via JwtStrategy
  async getConversations(
    @Headers('x-credential-id') credentialId: string,
    @Request() req,             // req.user = JWT payload from JwtStrategy.validate()
  ) {
    // All data is scoped to this specific WhatsApp credential (= one company)
    return this.service.getForCredential(credentialId, req.user);
  }
}

Gateway Middleware Verification

Beyond JWT validation, backends verify the HMAC signature.

Gateway middleware — NestJS 
const timestamp  = req.headers['x-gateway-timestamp'] as string;
const signature  = req.headers['x-gateway-signature'] as string;

// Replay attack: reject if timestamp older than 5 minutes
if (Date.now() - parseInt(timestamp) > 300_000) throw new UnauthorizedException();

const expected = createHmac('sha256', process.env.GATEWAY_SHARED_SECRET)
  .update(`${timestamp}:${req.method}:${req.path}`)
  .digest('hex');

if (signature !== expected) throw new UnauthorizedException();
